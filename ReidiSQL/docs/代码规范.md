## 1. 通用规范

### 1.1 命名规范

#### 文件命名
- **TypeScript/JavaScript**：camelCase 或 PascalCase
  - 组件文件：`PascalCase.tsx`（如 `Button.tsx`）
  - 工具文件：`camelCase.ts`（如 `dateUtils.ts`）
  - 测试文件：`*.test.ts` 或 `*.spec.ts`

- **Rust**：snake_case
  - 模块文件：`snake_case.rs`
  - 测试文件：内置 `#[cfg(test)]` 模块

#### 变量和函数命名
```typescript
// ✅ 推荐
const userName = 'John';
function getUserById(id: number) { }
const MAX_RETRY_COUNT = 3;

// ❌ 避免
const user_name = 'John';  // 不使用 snake_case
function GetUserById(id: number) { }  // 函数不用 PascalCase
const maxRetryCount = 3;  // 常量用 UPPER_SNAKE_CASE
```

### 1.2 代码格式

- **缩进**：2 个空格（不使用 Tab）
- **行宽**：100 字符
- **分号**：使用分号
- **引号**：单引号（HTML/JSX 中使用双引号）
- **尾随逗号**：ES5 模式（多行时添加）

### 1.3 注释规范

```typescript
/**
 * 查询数据库并返回结果
 *

  query: string,
```

## 2. TypeScript 规范

### 2.1 类型定义

#### 优先使用 interface 而不是 type
```typescript
// ✅ 推荐
interface User {
  id: number;
  name: string;
  email: string;
}

// ❌ 避免
type User = {
  id: number;
  name: string;
  email: string;
};
```

**例外**：联合类型、交叉类型使用 `type`
```typescript
type Status = 'pending' | 'active' | 'inactive';
type UserWithRole = User & { role: Role };
```

#### 避免使用 any
```typescript
// ❌ 避免
function process(data: any) { }

// ✅ 推荐
function process(data: unknown) {
  if (typeof data === 'object' && data !== null) {
    // 类型收窄
  }
}
```

### 2.2 可选链和空值合并

```typescript
// ✅ 推荐
const userName = user?.profile?.name ?? 'Anonymous';

// ❌ 避免
const userName = user && user.profile && user.profile.name 
  ? user.profile.name 
  : 'Anonymous';
```

### 2.3 枚举使用

```typescript
// ✅ 推荐：使用 const enum 或字符串字面量
type DatabaseType = 'mysql' | 'postgresql' | 'sqlite';

// 或使用 const enum
const enum ConnectionStatus {
  Disconnected = 'disconnected',
  Connecting = 'connecting',
  Connected = 'connected',
}

// ❌ 避免：数字枚举
enum ConnectionStatus {
  Disconnected,  // 0
  Connecting,    // 1
  Connected,     // 2
}
```

### 2.4 导入顺序

```typescript
// 1. 外部依赖
import React from 'react';
import { useState } from 'react';
import { Button } from 'antd';

// 2. 内部模块（按路径层级排序）
import { useConnection } from '@/hooks';
import { ConnectionService } from '@/services';
import { Connection } from '@/types';

// 3. 相对路径导入
import { formatQuery } from './utils';

```

## 3. React 规范

### 3.1 组件结构

```typescript
import React, { useState, useEffect } from 'react';
import { Button, Input } from 'antd';
import styles from './UserForm.module.css';

interface UserFormProps {
  initialData?: User;
  onSubmit: (data: User) => void;
  onCancel: () => void;
}

/**
 * 用户表单组件
 */
export const UserForm: React.FC<UserFormProps> = ({
  initialData,
  onSubmit,
  onCancel,
}) => {
  // 1. Hooks（按顺序）
  const [name, setName] = useState(initialData?.name ?? '');
  const [email, setEmail] = useState(initialData?.email ?? '');
  
  // 2. Effects
  useEffect(() => {
    // 副作用逻辑
  }, [name, email]);
  
  // 3. Event handlers
  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    onSubmit({ name, email });
  };
  
  const handleCancel = () => {
    onCancel();
  };
  
  // 4. Render
  return (
    <form className={styles.form} onSubmit={handleSubmit}>
      <Input value={name} onChange={(e) => setName(e.target.value)} />
      <Input value={email} onChange={(e) => setEmail(e.target.value)} />
      <div className={styles.actions}>
        <Button onClick={handleCancel}>取消</Button>
        <Button type="primary" htmlType="submit">提交</Button>
      </div>
    </form>
  );
};
```

### 3.3 状态管理

#### 使用 Zustand 进行全局状态管理

```typescript
import { create } from 'zustand';
import { Connection } from '@/types';

interface ConnectionStore {
  connections: Connection[];
  activeConnection: Connection | null;
  addConnection: (connection: Connection) => void;
  removeConnection: (id: string) => void;
  setActiveConnection: (id: string) => void;
}

export const useConnectionStore = create<ConnectionStore>((set) => ({
  connections: [],
  activeConnection: null,
  
  addConnection: (connection) =>
    set((state) => ({
      connections: [...state.connections, connection],
    })),
  
  removeConnection: (id) =>
    set((state) => ({
      connections: state.connections.filter((c) => c.id !== id),
    })),
  
  setActiveConnection: (id) =>
    set((state) => ({
      activeConnection: state.connections.find((c) => c.id === id) ?? null,
    })),
}));
```

#### 使用 React Query 管理服务端状态

```typescript

}
```

## 5. Node.js 规范

### 5.1 错误处理

```typescript
// 自定义错误类
export class DatabaseError extends Error {
  constructor(
    message: string,
    public code: string,
    public originalError?: Error
  ) {
    super(message);
    this.name = 'DatabaseError';
  }
}

// 统一的错误处理中间件
export const errorHandler: ErrorRequestHandler = (err, req, res, next) => {
  console.error('[Error]', err);
  
  if (err instanceof DatabaseError) {
    return res.status(400).json({
      error: err.message,
      code: err.code,
    });
  }
  
  res.status(500).json({
    error: 'Internal Server Error',
  });
};
```

### 5.2 异步编程

```typescript
// ✅ 推荐：使用 async/await
async function getUser(id: number): Promise<User> {
  const user = await db.query('SELECT * FROM users WHERE id = ?', [id]);
  if (!user) {
    throw new NotFoundError('User not found');
  }
  return user;
}

// ❌ 避免：Promise 链
function getUser(id: number): Promise<User> {
  return db.query('SELECT * FROM users WHERE id = ?', [id])
    .then(user => {
      if (!user) {
        throw new NotFoundError('User not found');
      }
      return user;
    });
}
```

### 6.2 错误处理

```rust
use thiserror::Error;

#[derive(Error, Debug)]
pub enum DatabaseError {
    #[error("Connection failed: {0}")]
    ConnectionFailed(#[from] std::io::Error),
    
    #[error("Query failed: {message}")]
    QueryFailed { message: String },
    
    #[error("Record not found: {id}")]
    NotFound { id: i64 },
}

pub type Result<T> = std::result::Result<T, DatabaseError>;

// 使用
fn get_user(id: i64) -> Result<User> {
    let user = query_user(id).map_err(|e| DatabaseError::QueryFailed {
        message: e.to_string(),
    })?;
    
    user.ok_or(DatabaseError::NotFound { id })
}
```

```typescript
import { describe, it, expect, beforeEach } from 'vitest';
import { formatQuery } from './queryFormatter';

describe('formatQuery', () => {
  beforeEach(() => {
    // 每个测试前执行
  });
  
  it('应该格式化 SELECT 语句', () => {
    const input = 'select * from users where id=1';
    const expected = 'SELECT * FROM users WHERE id = 1';
    
    expect(formatQuery(input)).toBe(expected);
  });
  
  it('应该处理空字符串', () => {
    expect(formatQuery('')).toBe('');
  });
  
  it('应该处理 null 值', () => {

  });
});
```

## 8. Git 规范

### 8.1 分支命名

```
feature/功能名称      # 新功能
bugfix/问题描述      # Bug 修复
hotfix/紧急修复      # 紧急热修复
release/版本号       # 发布分支
```

### 8.2 提交消息格式

遵循 [Conventional Commits](https://www.conventionalcommits.org/) 规范：

```
<type>[optional scope]: <description>

[optional body]

[optional footer(s)]
```

**Type 类型**：
- `feat`: 新功能
- `fix`: Bug 修复
- `docs`: 文档更新
- `style`: 代码格式（不影响代码运行）
- `refactor`: 代码重构
- `perf`: 性能优化
- `test`: 测试相关
- `build`: 构建系统或外部依赖
- `ci`: CI 配置文件
- `chore`: 其他修改

**示例**：
```bash
# ✅ 推荐
feat(editor): add SQL syntax highlighting
fix(connection): handle SSH tunnel timeout
docs(readme): update installation guide
refactor(database): extract driver interface

# ❌ 避免
update code
fix bug
changes
```
