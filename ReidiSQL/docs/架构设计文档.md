## 1. 架构概述

### 1.1 设计目标

ReidiSQL 的架构设计遵循以下核心原则：

1. **跨平台**：一套代码运行在 Windows、macOS、Linux
2. **高性能**：流畅处理百万级数据展示
3. **可扩展**：插件化架构，易于扩展新功能
4. **安全性**：保护用户数据和连接信息
5. **可维护**：模块化设计，清晰的代码组织

### 1.2 架构演进

#### 原架构（HeidiSQL - Delphi/VCL）
```
┌─────────────────────────────┐
│   Windows VCL UI Layer     │
├─────────────────────────────┤
│   Business Logic (Pascal)  │
├─────────────────────────────┤
│   FireDAC Data Access      │
├─────────────────────────────┤
│   Database Drivers (DLL)   │
└─────────────────────────────┘
```

**限制**：
- 仅支持 Windows
- 依赖商业组件（FireDAC）
- 单进程架构，性能受限
- UI 框架过时

#### 新架构（ReidiSQL - Tauri + React + Node.js）
```
┌──────────────────────────────────────────┐
│         React Frontend (Web)             │
│  UI Components + State Management        │
└──────────────┬───────────────────────────┘
               │ IPC (Tauri Commands)
┌──────────────▼───────────────────────────┐
│        Tauri Backend (Rust)              │
│  System APIs + Process Management        │
└──────────────┬───────────────────────────┘
               │ HTTP/WebSocket (localhost)
┌──────────────▼───────────────────────────┐
│      Node.js Service (TypeScript)        │
│  Database Drivers + Query Engine         │
└──────────────┬───────────────────────────┘
               │ Database Protocols
┌──────────────▼───────────────────────────┐
│         Database Servers                 │
│  MySQL / PostgreSQL / SQLite / ...       │
└──────────────────────────────────────────┘
```

**优势**：
- 真正的跨平台支持
- 使用开源技术栈
- 多进程架构，性能更好
- 现代化的 UI/UX

#### 4.1.1 架构设计

```typescript
// 统一驱动接口
interface DatabaseDriver {
  connect(config: ConnectionConfig): Promise<Connection>;
  disconnect(connectionId: string): Promise<void>;
  executeQuery(connectionId: string, query: string, params?: any[]): Promise<QueryResult>;
  getMetadata(connectionId: string): Promise<DatabaseMetadata>;
  startTransaction(connectionId: string): Promise<string>;
  commitTransaction(transactionId: string): Promise<void>;
  rollbackTransaction(transactionId: string): Promise<void>;
}

// MySQL 驱动实现
class MySQLDriver implements DatabaseDriver {
  private pool: Pool;
  
  async connect(config: ConnectionConfig): Promise<Connection> {
    this.pool = createPool({
      host: config.host,
      port: config.port,
      user: config.username,
      password: config.password,
      database: config.database,
    });
    
    const connection = await this.pool.getConnection();
    return {
      id: generateId(),
      driver: 'mysql',
      connection,
    };
  }
  
  async executeQuery(connectionId: string, query: string, params?: any[]): Promise<QueryResult> {
    const connection = this.getConnection(connectionId);
    const [rows, fields] = await connection.execute(query, params);
    
    return {
      rows: rows as any[],
      columns: fields.map(f => ({
        name: f.name,
        type: this.mapColumnType(f.type),
      })),
      rowCount: rows.length,
    };
  }
}
```

#### 4.1.2 连接管理

```typescript
class ConnectionManager {
  private connections: Map<string, Connection> = new Map();
  private drivers: Map<string, DatabaseDriver> = new Map();
  
  // 注册驱动
  registerDriver(type: string, driver: DatabaseDriver): void {
    this.drivers.set(type, driver);
  }
  
  // 创建连接
  async createConnection(config: ConnectionConfig): Promise<string> {
    const driver = this.drivers.get(config.type);
    if (!driver) {
      throw new Error(`Unsupported database type: ${config.type}`);
    }
    
    const connection = await driver.connect(config);
    this.connections.set(connection.id, connection);
    
    return connection.id;
  }
  
  // 关闭连接
  async closeConnection(connectionId: string): Promise<void> {
    const connection = this.connections.get(connectionId);
    if (connection) {
      await connection.driver.disconnect(connectionId);
      this.connections.delete(connectionId);
    }
  }
}
```

### 4.2 SQL 编辑器模块

#### 4.2.1 Monaco Editor 集成

```typescript
import * as monaco from 'monaco-editor';

class SQLEditor {
  private editor: monaco.editor.IStandaloneCodeEditor;
  private completionProvider: SQLCompletionProvider;
  
  constructor(container: HTMLElement, options: EditorOptions) {
    this.editor = monaco.editor.create(container, {
      value: '',
      language: 'sql',
      theme: 'vs-dark',
      automaticLayout: true,
      minimap: { enabled: false },
      fontSize: 14,
      wordWrap: 'on',
      ...options,
    });
    
    this.registerLanguage();
    this.registerCompletionProvider();
  }
  
  private registerLanguage(): void {
    monaco.languages.register({ id: 'sql' });
    
    monaco.languages.setMonarchTokensProvider('sql', {
      keywords: [
        'SELECT', 'FROM', 'WHERE', 'INSERT', 'UPDATE', 'DELETE',
        'CREATE', 'DROP', 'ALTER', 'JOIN', 'LEFT', 'RIGHT',
        // ... 更多关键字
      ],
      
      operators: [
        '=', '>', '<', '!=', '<=', '>=',
        // ... 更多操作符
      ],
      
      // 语法规则
      tokenizer: {
        root: [
          [/[a-zA-Z_]\w*/, { cases: { '@keywords': 'keyword', '@default': 'identifier' } }],
          [/[<>!=]=?/, 'operator'],
          [/[0-9]+/, 'number'],
          [/'/, 'string', '@string'],
          // ... 更多规则
        ],
        
        string: [
          [/[^']+/, 'string'],
          [/'/, 'string', '@pop'],
        ],
      },
    });
  }
  
  private registerCompletionProvider(): void {
    this.completionProvider = new SQLCompletionProvider();
    
    monaco.languages.registerCompletionItemProvider('sql', {
      triggerCharacters: ['.', ' ', '('],
      provideCompletionItems: async (model, position) => {
        const word = model.getWordUntilPosition(position);
        const suggestions = await this.completionProvider.getSuggestions({
          word: word.word,
          position,
          model,
        });
        
        return { suggestions };
      },
    });
  }
}
```

#### 4.2.2 智能提示引擎

```typescript
interface CompletionContext {
  word: string;
  position: monaco.Position;
  model: monaco.editor.ITextModel;
}

class SQLCompletionProvider {
  private metadataCache: MetadataCache;
  
  async getSuggestions(context: CompletionContext): Promise<monaco.languages.CompletionList> {
    const suggestions: monaco.languages.CompletionItem[] = [];
    
    // 1. SQL 关键字
    suggestions.push(...this.getKeywordSuggestions(context.word));
    
    // 2. 表名
    suggestions.push(...await this.getTableSuggestions(context.word));
    
    // 3. 列名（根据上下文）
    if (this.isInSelectClause(context)) {
      suggestions.push(...await this.getColumnSuggestions(context));
    }
    
    // 4. 函数名
    suggestions.push(...this.getFunctionSuggestions(context.word));
    
    // 5. 查询历史
    suggestions.push(...this.getHistorySuggestions(context.word));
    
    return { suggestions };
  }
  
  private async getTableSuggestions(prefix: string): Promise<monaco.languages.CompletionItem[]> {
    const tables = await this.metadataCache.getTables();
    
    return tables
      .filter(t => t.name.toLowerCase().startsWith(prefix.toLowerCase()))
      .map(table => ({
        label: table.name,
        kind: monaco.languages.CompletionItemKind.Class,
        insertText: table.name,
        detail: `Table in ${table.schema}`,
      }));
  }
}
```

```typescript
import { AgGridReact } from 'ag-grid-react';
import { ColDef, GridReadyEvent } from 'ag-grid-community';

interface DataGridProps {

}

export const DataGrid: React.FC<DataGridProps> = ({
  connectionId,
  query,
  pageSize = 1000,
}) => {
  const [rowData, setRowData] = useState<any[]>([]);
  const [columnDefs, setColumnDefs] = useState<ColDef[]>([]);
  const gridRef = useRef<AgGridReact>(null);
  
  // 流式加载数据
  const loadData = async () => {
    const response = await fetch(`/api/query/execute`, {
      method: 'POST',
      body: JSON.stringify({
        connectionId,
        query,
        stream: true,
      }),
    });
    
    const reader = response.body?.getReader();
    const decoder = new TextDecoder();
    
    while (true) {
      const { done, value } = await reader!.read();
      if (done) break;
      
      const chunk = JSON.parse(decoder.decode(value));
      setRowData(prev => [...prev, ...chunk.rows]);
    }
  };
  
  return (
    <AgGridReact
      ref={gridRef}
      rowData={rowData}
      columnDefs={columnDefs}
      rowBuffer={10}
      suppressRowVirtualisation={false}
      pagination={true}
      paginationPageSize={pageSize}
      enableCellTextSelection={true}
    />
  );
};
```

### 4.4 SSH 隧道模块

```typescript
import { Client } from 'ssh2';

class SSHTunnelManager {
  private client: Client | null = null;
  private server: any = null;
  
  async connect(config: SSHConfig): Promise<number> {
    return new Promise((resolve, reject) => {
      this.client = new Client();
      
      this.client.on('ready', async () => {
        try {
          // 创建本地 SOCKS 代理
          const localPort = await this.startForward(config);
          resolve(localPort);
        } catch (error) {
          reject(error);
        }
      });
      
      this.client.on('error', reject);
      
      this.client.connect({
        host: config.host,
        port: config.port,
        username: config.username,
        password: config.password,
        privateKey: config.privateKey,
      });
    });
  }
  
  private async startForward(config: SSHConfig): Promise<number> {
    return new Promise((resolve, reject) => {
      const server = net.createServer((socket) => {
        this.client!.forwardOut(
          '127.0.0.1',
          0,
          config.remoteHost,
          config.remotePort,
          (err, stream) => {
            if (err) {
              socket.destroy();
              return;
            }
            
            socket.pipe(stream).pipe(socket);
          }
        );
      });
      
      server.listen(0, '127.0.0.1', () => {
        const address = server.address();
        const port = typeof address === 'object' ? address!.port : 0;
        this.server = server;
        resolve(port);
      });
    });
  }
  
  async disconnect(): Promise<void> {
    if (this.server) {
      this.server.close();
    }
    if (this.client) {
      this.client.end();
    }
  }
}
```

### 7.1 连接池

```typescript
import { createPool } from 'mysql2/promise';

class ConnectionPoolManager {
  private pools: Map<string, Pool> = new Map();
  
  createPool(config: ConnectionConfig): Pool {
    const pool = createPool({
      host: config.host,
      port: config.port,
      user: config.username,
      password: config.password,
      database: config.database,
      connectionLimit: 10,
      queueLimit: 100,
      waitForConnections: true,
    });
    
    this.pools.set(config.id, pool);
    return pool;
  }
  
  // 连接预热
  async warmup(connectionId: string): Promise<void> {
    const pool = this.pools.get(connectionId);
    if (!pool) return;
    
    // 预创建连接
    const connections = await Promise.all(
      Array(5).fill(null).map(() => pool.getConnection())
    );
    
    // 立即释放回池
    connections.forEach(conn => conn.release());
  }
}
```

### 7.2 缓存策略

```typescript
import { LRUCache } from 'lru-cache';

class MetadataCache {
  private cache: LRUCache<string, any>;
  
  constructor() {
    this.cache = new LRUCache({
      max: 500, // 最多缓存 500 个条目
      ttl: 5 * 60 * 1000, // 5 分钟过期
    });
  }
  
  // 缓存表结构
  setTableStructure(connectionId: string, tableName: string, columns: Column[]) {
    const key = `${connectionId}:${tableName}`;
    this.cache.set(key, columns);
  }
  
  // 获取表结构
  getTableStructure(connectionId: string, tableName: string): Column[] | undefined {
    const key = `${connectionId}:${tableName}`;
    return this.cache.get(key);
  }
  
  // 清除缓存
  invalidate(connectionId: string, tableName?: string) {
    if (tableName) {
      const key = `${connectionId}:${tableName}`;
      this.cache.delete(key);
    } else {
      // 清除该连接的所有缓存
      for (const key of this.cache.keys()) {
        if (key.startsWith(connectionId)) {
          this.cache.delete(key);
        }
      }
    }
  }
}
```

### 7.3 流式查询

```typescript
// 后端：流式返回结果
async function executeStreamQuery(
  req: Request,
  res: Response
): Promise<void> {
  const { connectionId, query } = req.body;
  
  const connection = await pool.getConnection();
  const stream = connection.query(query).stream();
  
  res.setHeader('Content-Type', 'application/json');
  res.setHeader('Transfer-Encoding', 'chunked');
  
  let rowCount = 0;
  const batchSize = 100;
  let batch: any[] = [];
  
  stream.on('data', (row: any) => {
    batch.push(row);
    rowCount++;
    
    if (batch.length >= batchSize) {
      res.write(JSON.stringify({ rows: batch }) + '\n');
      batch = [];
    }
  });
  
  stream.on('end', () => {
    if (batch.length > 0) {
      res.write(JSON.stringify({ rows: batch }) + '\n');
    }
    res.write(JSON.stringify({ done: true, rowCount }) + '\n');
    res.end();
  });
}
```

### 10.2 性能监控

```typescript
// 查询性能监控
class QueryPerformanceMonitor {
  private metrics: Map<string, number[]> = new Map();
  
  recordQuery(connectionId: string, duration: number): void {
    if (!this.metrics.has(connectionId)) {
      this.metrics.set(connectionId, []);
    }
    
    this.metrics.get(connectionId)!.push(duration);
    
    // 保持最近 100 次查询
    if (this.metrics.get(connectionId)!.length > 100) {
      this.metrics.get(connectionId)!.shift();
    }
  }
  
  getAverageDuration(connectionId: string): number {
    const durations = this.metrics.get(connectionId) || [];
    if (durations.length === 0) return 0;
    
    const sum = durations.reduce((a, b) => a + b, 0);
    return sum / durations.length;
  }
  
  getSlowQueries(threshold: number = 1000): Array<{ connectionId: string, avgDuration: number }> {
    const slow: Array<{ connectionId: string, avgDuration: number }> = [];
    
    for (const [connectionId, durations] of this.metrics) {
      const avg = durations.reduce((a, b) => a + b, 0) / durations.length;
      if (avg > threshold) {
        slow.push({ connectionId, avgDuration: avg });
      }
    }
    
    return slow;
  }
}
```
